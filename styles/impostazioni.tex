% Impostazioni di pagina e macro per il frontespizio e la dedica della tesi.
% Va incluso nel preambolo del documento principale con \input{styles/impostazioni}.

% Formato pagina fisico: B5 ISO (176mm x 250mm).
% Margini: interno 3cm, esterno 2cm; intestazione e piè di pagina 2cm
% (includehead/includefoot fanno rientrare intestazione e piè di pagina
% nei rispettivi margini di 2cm, invece di aggiungersi ad essi).
\usepackage[b5paper, includehead, includefoot, inner=3cm, outer=2cm, top=2cm, bottom=2cm]{geometry}
\setlength{\parindent}{15pt}
\setcounter{tocdepth}{5}  %serve per mettere nell'indice i sotto-sotto paragrafi
\setcounter{secnumdepth}{5} %serve per numerare i sotto-sotto paragrafi

% Macro di supporto per \frontespizio: data una lista di nomi separati da virgola,
% conta quanti sono e li ricompone separati da \\ (una riga per nome).
% #1 lista, #2 macro che riceverà il conteggio, #3 macro che riceverà il testo
\makeatletter
\newcommand{\listaContaEStampa}[3]{%
    \begingroup
    \count@=0\relax
    \def\@listaTesto{}%
    \@for\@listaElemento:=#1\do{%
        \advance\count@ by1\relax
        \ifnum\count@=1
            \xdef\@listaTesto{\@listaElemento}%
        \else
            \xdef\@listaTesto{\@listaTesto\\\@listaElemento}%
        \fi
    }%
    \xdef#2{\the\count@}%
    \xdef#3{\@listaTesto}%
    \endgroup
}
\makeatother

% Macro per generare il frontespizio della tesi.
% Parametri:
%   #1 Dipartimento (es. "Ingegneria Civile, Energia, Ambiente e Materiali")
%   #2 Corso di Laurea (es. "Ingegneria Informatica")
%   #3 Titolo della tesi
%   #4 Candidato/i (per più candidati separare i nomi con una virgola)
%   #5 Relatore/i (per più relatori separare i nomi con una virgola)
%   #6 Correlatore/i, lasciare vuoto {} se assente (separare con una virgola se più di uno)
%   #7 Anno Accademico (es. "2023-2024")
% La dicitura "Relatore/Correlatore/Candidato" si trasforma automaticamente al
% plurale quando vengono indicati più nomi.
\newcommand{\frontespizio}[7]{%
    \begin{titlepage}
        \begin{center}

        \large{\bf Università degli Studi Mediterranea di Reggio Calabria}

        \vspace*{1mm}

        \large{Dipartimento di #1}

        \vspace*{1mm}

        \normalsize{Corso di Laurea in #2}

        \vspace*{1mm}

        \hspace*{-0mm}

        \rule{125mm}{.2mm} %{lunghezza}{spessore}

        \vspace{18mm}

        \begin{figure}[h!]
            \centerline{\includegraphics[width=3cm]{images/logounirc.png}}
        \end{figure}

        \vspace{5mm}

        \textbf{Tesi di Laurea}

        \vspace{5mm}

        %% TITOLO DELLA TESI
        \large{\bf #3}

        \vspace{22mm}

        \listaContaEStampa{#5}{\NumRelatori}{\TestoRelatori}%
        \listaContaEStampa{#6}{\NumCorrelatori}{\TestoCorrelatori}%
        \listaContaEStampa{#4}{\NumCandidati}{\TestoCandidati}%

        \noindent
        \begin{minipage}[t]{5cm}
            {\large \ifnum\NumRelatori>1 Relatori\else Relatore\fi}\\[2mm]
            \TestoRelatori\\[6mm]
            \ifnum\NumCorrelatori>0
                {\large \ifnum\NumCorrelatori>1 Correlatori\else Correlatore\fi}\\[2mm]
                \TestoCorrelatori
            \fi
        \end{minipage}
        \hfill
        \begin{minipage}[t]{5cm}
            {\large \ifnum\NumCandidati>1 Candidati\else Candidato\fi}\\[2mm]
            \TestoCandidati
        \end{minipage}

        \rule{125mm}{.2mm}

        \textbf{Anno Accademico #7}
        \end{center}
    \end{titlepage}
}

% Macro per generare la pagina di dedica.
% #1 Testo della dedica
\newcommand{\dedica}[1]{%
    \newpage
    \newpage
    \cleardoublepage
    \thispagestyle{empty}
    \vspace*{\stretch{1}}
    \begin{flushright}
    \itshape
    #1
    \end{flushright}
    \vspace{\stretch{2}}
    \cleardoublepage
}
